async function generateQuizWithAI() {
    const text = document.getElementById('raw-text-input').value.trim();
    const apiKey = localStorage.getItem('gemini_api_key');

    if (!text) return alert("Error: The text box is empty! Please paste text or upload a PDF with readable text.");
    if (!apiKey) {
        openApiKeyModal();
        return alert("Please enter a free Gemini API Key to use AI Generation.");
    }

    const statusEl = document.getElementById('upload-status');
    document.getElementById('upload-status-text').textContent = "AI Agent is generating exam questions...";
    statusEl.classList.remove('hidden');

    try {
        const prompt = `Based on this study text, generate 5 multiple-choice exam questions. Return strictly valid JSON array with format:
[
  {
    "question": "Question string",
    "options": ["Opt A", "Opt B", "Opt C", "Opt D"],
    "correctIndex": 0,
    "hint": "Brief hint",
    "explanation": "Detailed answer explanation"
  }
]
Text: ${text.substring(0, 12000)}`;

        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
        });

        const data = await response.json();

        // Check if Google returned an API error
        if (data.error) {
            throw new Error(`Google API Error: ${data.error.message} (Code ${data.error.code})`);
        }

        if (!data.candidates || !data.candidates[0]) {
            throw new Error("No response generated by AI. The text might have triggered safety filters.");
        }

        let rawRes = data.candidates[0].content.parts[0].text;
        rawRes = rawRes.replace(/```json/g, '').replace(/```/g, '').trim();
        const questions = JSON.parse(rawRes);

        const newSet = {
            id: 'set-' + Date.now(),
            title: document.getElementById('custom-set-title').value || "AI Quiz Set",
            questions: questions
        };

        quizSets.push(newSet);
        populateQuizSelector();
        currentSetId = newSet.id;
        document.getElementById('quiz-selector').value = currentSetId;
        loadSelectedQuizSet();
        switchTab('quiz');
    } catch (err) {
        console.error(err);
        alert(`Generation Failed:\n${err.message}`);
    } finally {
        statusEl.classList.add('hidden');
    }
}
